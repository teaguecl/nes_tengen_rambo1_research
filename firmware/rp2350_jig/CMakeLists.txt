# RAMBO-1 RP2350 jig firmware. Build: set PICO_SDK_PATH (Pico SDK 2.x) and the Arm GNU toolchain in PATH, then: cmake -S . -B build && cmake --build build -j
cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR})
set(PICO_BOARD rambo_bb48)
include($ENV{PICO_SDK_PATH}/pico_sdk_init.cmake)
project(rp2350_jig C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(rp2350_jig main.c engine.c)
target_link_libraries(rp2350_jig pico_stdlib hardware_pio hardware_dma hardware_gpio)
pico_generate_pio_header(rp2350_jig ${CMAKE_CURRENT_LIST_DIR}/engine.pio)
pico_enable_stdio_usb(rp2350_jig 1)
pico_enable_stdio_uart(rp2350_jig 0)
pico_add_extra_outputs(rp2350_jig)
